cmake_minimum_required(VERSION 3.24)
project(bvh_tiled_geometry LANGUAGES CXX CUDA)
find_package(Python3 3.12 EXACT REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(CUDAToolkit REQUIRED)
if(NOT DEFINED CODE_ROOT)
    message(FATAL_ERROR "Pass an explicit frozen projection directory")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
add_library(gnss_gpu_bvh STATIC "${CODE_ROOT}/src/raytrace/bvh.cu")
target_include_directories(gnss_gpu_bvh PUBLIC "${CODE_ROOT}/include")
target_link_libraries(gnss_gpu_bvh CUDA::cudart)
set_target_properties(gnss_gpu_bvh PROPERTIES CUDA_ARCHITECTURES 75 POSITION_INDEPENDENT_CODE ON)
pybind11_add_module(_bvh "${CODE_ROOT}/python/gnss_gpu/_bvh_bindings.cpp")
target_link_libraries(_bvh PRIVATE gnss_gpu_bvh)
