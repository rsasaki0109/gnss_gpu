cmake_minimum_required(VERSION 3.18)
project(rbpf_likelihood_probe LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
find_package(CUDAToolkit REQUIRED)
add_library(rbpf_likelihood_probe SHARED probe.cu)
set_target_properties(rbpf_likelihood_probe PROPERTIES CUDA_ARCHITECTURES 75)
target_link_libraries(rbpf_likelihood_probe PRIVATE CUDA::cudart)
target_compile_options(rbpf_likelihood_probe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ptxas-options=-v>)
