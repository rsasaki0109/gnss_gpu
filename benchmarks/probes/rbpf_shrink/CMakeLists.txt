cmake_minimum_required(VERSION 3.24)
project(rbpf_shrink LANGUAGES CXX)
find_package(Python3 3.12 EXACT REQUIRED COMPONENTS Interpreter Development.Module)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This standalone probe requires 64-bit Windows CPython 3.12")
endif()
Python3_add_library(_rbpf_shrink_probe MODULE WITH_SOABI shrink.cpp)
target_compile_features(_rbpf_shrink_probe PRIVATE cxx_std_17)
target_compile_options(_rbpf_shrink_probe PRIVATE /O2 /fp:strict /W4)
