cmake_minimum_required(VERSION 3.24)
project(rbpf_dd_pack LANGUAGES CXX)
find_package(Python3 3.12 EXACT REQUIRED COMPONENTS Interpreter Development)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This standalone probe requires 64-bit Windows CPython 3.12")
endif()
add_library(rbpf_dd_pack SHARED pack.cpp)
target_link_libraries(rbpf_dd_pack PRIVATE Python3::Python)
target_compile_features(rbpf_dd_pack PRIVATE cxx_std_17)
target_compile_options(rbpf_dd_pack PRIVATE /O2 /fp:strict /W4)
